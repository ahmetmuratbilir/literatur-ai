1. Unpaywall PDF Çözücü Servisi
Öncelik: 1

Herhangi bir makalenin DOI numarasını alır, dünyanın 50.000+ açık arşivini tarar ve kullanıcıya doğrudan indirilebilir açık erişim PDF linkini döner.

Hedef Dosya:server/services/unpaywall.js
Uç Nokta:https://api.unpaywall.org/v2/{doi}
Maliyet:%100 Ücretsiz
server/services/unpaywall.js
import axios from 'axios';

/**
 * Bir makalenin DOI'si ile yasal açık erişim PDF URL'sini bulur.
 */
export async function getOpenAccessPdf(doi) {
  if (!doi) return null;
  try {
    const email = process.env.CONTACT_EMAIL || 'info@literatur-ai.com';
    const cleanDoi = encodeURIComponent(doi.trim());
    const res = await axios.get(`https://api.unpaywall.org/v2/${cleanDoi}?email=${email}`, { timeout: 3500 });

    if (res.data?.is_oa && res.data?.best_oa_location?.url_for_pdf) {
      return {
        isOa: true,
        pdfUrl: res.data.best_oa_location.url_for_pdf,
        landingPage: res.data.best_oa_location.url_for_landing_page,
        license: res.data.best_oa_location.license
      };
    }
    return null;
  } catch (err) {
    // Unpaywall 404 dönse bile ana akışı kesmez
    return null;
  }
}
2. PubMed (NCBI Entrez) Arama Servisi
Öncelik: 2

ABD Ulusal Tıp Kütüphanesi'ndeki 36 Milyon klinik, genetik ve tıp araştırmasını taranabilir kılar.

Hedef Dosya:server/services/pubmed.js
Uç Nokta:https://eutils.ncbi.nlm.nih.gov/entrez/eutils/
Maliyet:%100 Ücretsiz
server/services/pubmed.js
import axios from 'axios';

export async function searchPubMed(query, count = 10) {
  if (!query?.trim()) return { results: [], totalFound: 0 };
  try {
    // 1. Arama yapıp PubMed ID (PMID) listesini alıyoruz
    const searchRes = await axios.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi', {
      params: { db: 'pubmed', term: query, retmode: 'json', retmax: count },
      timeout: 5000
    });
    
    const ids = searchRes.data?.esearchresult?.idlist || [];
    const total = parseInt(searchRes.data?.esearchresult?.count || 0, 10);
    if (ids.length === 0) return { results: [], totalFound: 0 };

    // 2. Detaylı bibliyografik verileri çekiyoruz (esummary)
    const sumRes = await axios.get('https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi', {
      params: { db: 'pubmed', id: ids.join(','), retmode: 'json' },
      timeout: 5000
    });

    const resultDict = sumRes.data?.result || {};
    const results = ids.map(id => {
      const item = resultDict[id];
      if (!item) return null;
      const doi = item.articleids?.find(a => a.idtype === 'doi')?.value || '';
      return {
        id: `pubmed-${id}`,
        title: item.title?.replace(/<[^>]*>/g, '') || 'Başlıksız Yayın',
        authors: (item.authors || []).map(a => a.name).join(', '),
        year: item.pubdate ? parseInt(item.pubdate.slice(0, 4), 10) : null,
        journal: item.source || 'PubMed Central',
        doi,
        url: doi ? `https://doi.org/${doi}` : `https://pubmed.ncbi.nlm.nih.gov/${id}/`,
        source: 'PubMed',
        citedBy: 0
      };
    }).filter(Boolean);

    return { results, totalFound: total };
  } catch (err) {
    return { results: [], totalFound: 0 };
  }
}

Öncelik: 3

Bir makaleye atıf veren diğer yayınları veya kaynakçayı JSON olarak çeker. Reader Drawer içinde "Bu makaleyi kaynak gösteren yayınlar" sekmesi oluşturur.

Hedef Dosya:server/services/openCitations.js
Uç Nokta:https://opencitations.net/index/coci/api/v1/citations/
Maliyet:%100 Ücretsiz
server/services/openCitations.js
import axios from 'axios';

export async function getCitationsForPaper(doi) {
  if (!doi) return [];
  try {
    const res = await axios.get(`https://opencitations.net/index/coci/api/v1/citations/${encodeURIComponent(doi)}`, { timeout: 4000 });
    return (res.data || []).map(item => ({
      citingDoi: item.citing,
      creationYear: item.creation ? item.creation.slice(0, 4) : null
    }));
  } catch (err) {
    return [];
  }
}

    return { results: [], totalFound: 0 };

}

⚡ 3. Sunucuya (server/index.js) Paralel Ekleme

🚀 Sıfır Gecikmeli Paralel Arama (Promise.allSettled)
Yeni servisleri server/index.js içindeki arama fonksiyonuna eklerken mevcut Promise.allSettled yapısını kullanırız. Böylece aramaya yeni 3 kaynak eklense bile cevap süresi 1 milisaniye bile uzamaz!
server/index.js — Paralel Arama Örneği
// 1. Yeni servisleri import et
import { searchPubMed } from './services/pubmed.js';
import { searchEuropePmc } from './services/europePmc.js';

// 2. Paralel sorgu havuzuna ekle:
const searchPromises = [
  searchScopus(query, count),
  searchOpenAlex(query, count),
  searchCrossref(query, count),
  searchCORE(query, count),
  searchArxiv(query, count),
  searchDOAJ(query, count),
  searchPubMed(query, count),    // 👈 YENİ TIP KAYNAĞI
  searchEuropePmc(query, count), // 👈 YENİ AVRUPA KAYNAĞI
];

const settled = await Promise.allSettled(searchPromises);
